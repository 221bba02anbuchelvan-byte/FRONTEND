Demonstrate global vs local scope. let globalVar = "I am Global"; // Accessible everywhere function scopeDemo() { let localVar = "I am Local"; // Accessible ONLY inside this function console.log(globalVar); // Works console.log(localVar); // Works } scopeDemo(); // console.log(localVar); // Error: localVar is not defined 6. Give example for callback Function function processName(name, callback) { console.log("Processing name: " + name); callback(); // Calling the callback function } function showDone() { console.log("Task Completed!"); } processName("Alex", showDone); 7. Give example for Function Closure function outerFunction() { let count = 0; // Outer scope variable return function innerFunction() { count++; // Accesses outer variable even after outerFunction has executed return count; }; } const counter = outerFunction(); console.log(counter()); // Output: 1 console.log(counter()); // Output: 2
