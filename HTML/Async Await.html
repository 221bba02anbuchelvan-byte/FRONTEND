Original Promise Code (.then / .catch) function fetchData() { return new Promise((resolve) => { setTimeout(() => resolve("Data fetched successfully!"), 1000); }); } // Handling with .then() fetchData() .then((data) => console.log(data)) .catch((error) => console.error(error)); 2. Rewritten Code using async/await function fetchData() { return new Promise((resolve) => { setTimeout(() => resolve("Data fetched successfully!"), 1000); }); } // Rewritten with async/await and try...catch async function getData() { try { const result = await fetchData(); console.log(result); } catch (error) { console.error(error); } } getData();
